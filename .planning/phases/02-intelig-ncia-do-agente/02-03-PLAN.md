  ```typescript
  function round1(n: number): number {
    return Math.round(n * 10) / 10;
  }
  function avg(arr: number[]): number {
    if (arr.length === 0) return 0;
    return arr.reduce((a, b) => a + b, 0) / arr.length;
  }

  export const selectBenchmarkAggregates = (
    sessions: Session[],
    filters: { promptVersion?: string; motivationLabel?: "BAIXA" | "MÉDIA" | "ALTA" } = {},
  ): { avgReductionPct: number | null; avgRounds: number | null; avgRating: number | null; count: number } => {
    const ended = sessions.filter(
      (s) =>
        s.status === "ended" &&
        s.finalPrice != null &&
        (filters.promptVersion == null || s.promptVersion === filters.promptVersion) &&
        (filters.motivationLabel == null || s.motivationScore?.label === filters.motivationLabel),
    );

    if (ended.length === 0) {
      return { avgReductionPct: null, avgRounds: null, avgRating: null, count: 0 };
    }

    const reductions = ended.map((s) => (1 - (s.finalPrice as number) / s.fipe) * 100);
    const rounds = ended.map((s) => s.round);
    const rated = ended.filter((s) => s.userRating != null);

    return {
      avgReductionPct: round1(avg(reductions)),
      avgRounds: round1(avg(rounds)),
      avgRating: rated.length > 0 ? round1(avg(rated.map((s) => s.userRating as number))) : null,
      count: ended.length,
    };
  };

  export const selectGatePass = (sessions: Session[]): { pass: boolean; reasons: string[] } => {
    const agg = selectBenchmarkAggregates(sessions, { promptVersion: "v2.0.0" });
    const reasons: string[] = [];
    if (agg.count < 20) reasons.push(`N=${agg.count} (precisa 20)`);
    if (agg.avgReductionPct != null && agg.avgReductionPct < 20)
      reasons.push(`redução ${agg.avgReductionPct}% (precisa ≥20%)`);
    if (agg.avgRating != null && agg.avgRating < 3.5)
      reasons.push(`qualidade ${agg.avgRating}/5 (precisa ≥3.5)`);
    const hardStopHits = sessions.filter((s) => s.flags?.walkAwayExceeded).length;
    if (hardStopHits > 0) reasons.push(`${hardStopHits} sessões com walkAway excedido`);
    return { pass: reasons.length === 0, reasons };
  };
  ```
