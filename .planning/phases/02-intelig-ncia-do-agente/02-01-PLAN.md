  ```typescript
  import { z } from "zod";
  import type { Listing } from "@/lib/schemas/listing";

  export const motivationScoreSchema = z.object({
    score: z.number().int().min(0).max(100),
    label: z.enum(["BAIXA", "MÉDIA", "ALTA"]),
    rationale: z.string().min(20).max(400),
  });
  export type MotivationScore = z.infer<typeof motivationScoreSchema>;

  const MOTIVO_WEIGHTS: Record<NonNullable<Listing["motivoDaVenda"]>, number> = {
    mudanca_cidade: 10,
    upgrade_veiculo: 5,
    necessidade_financeira: 20,
    outros: 0,
  };

  export function computeMotivationScore(listing: Listing): MotivationScore {
    // Pesos D-04: dias online × 0.6 cap 50; reduções × 10 cap 30; motivo mapping 0/5/10/20
    const diasOnlineScore = Math.min(50, listing.diasOnline * 0.6);
    const reducoesScore = Math.min(30, listing.reducoes * 10);
    const motivoScore = MOTIVO_WEIGHTS[listing.motivoDaVenda ?? "outros"];

    const score = Math.round(diasOnlineScore + reducoesScore + motivoScore);
    // Thresholds D-07: <40 BAIXA, 40-69 MÉDIA, >=70 ALTA
    const label: MotivationScore["label"] = score >= 70 ? "ALTA" : score >= 40 ? "MÉDIA" : "BAIXA";
    const rationale = `${listing.diasOnline} dias online, ${listing.reducoes} reduções, motivo "${listing.motivoDaVenda ?? "não informado"}".`;

    // Defense in depth — programming error se fórmula gerar shape inválido
    return motivationScoreSchema.parse({ score, label, rationale });
  }
  ```

    for (let i = 1; i < msgs.length; i++) {
      if (msgs[i].role === msgs[i - 1].role) return false;
