import Anthropic from "@anthropic-ai/sdk";
import type { StreamOptions } from "./types";

const DEFAULT_MODEL = "claude-sonnet-4-6";

export async function streamAnthropic(opts: StreamOptions): Promise<void> {
  const apiKey = process.env.ANTHROPIC_API_KEY;
  if (!apiKey) throw new Error("ANTHROPIC_API_KEY missing");
  const model = process.env.ANTHROPIC_MODEL ?? DEFAULT_MODEL;
  const client = new Anthropic({ apiKey });
  const stream = client.messages.stream(
    { model, max_tokens: opts.maxTokens, system: opts.systemPrompt, messages: opts.messages },
    { signal: opts.signal },
  );
  stream.on("text", (chunk: string) => opts.onText(chunk));
  await stream.finalMessage();
}

  ```typescript
  /**
   * Zustand persist migration v1 → v2 (D-14).
   *
   * Strategy (RESEARCH Opção Y): manter key "autoagent-playground-v1" + bump version 1→2.
   * Zero fallback logic for key rename — simpler, less room for data loss.
   *
   * Risk mitigation: NEVER throws. Malformed shape → return init state + log warn
   * via onRehydrateStorage in store config.
   */

  import type { AgentConfig, Session } from "@/lib/types/session";

  export const DEFAULT_AGENT_CONFIG: AgentConfig = {
    targetDiscount: 0.25,
    maxRounds: 6,
    tone: "casual",
    initialAnchorStrategy: "moderado",
  };

  // Init shape — returned when persisted is missing/malformed.
  function initState() {
    return {
      currentSession: null,
      history: [],
      agentConfig: DEFAULT_AGENT_CONFIG,
    };
  }

  interface PersistedV1 {
    currentSession: Partial<Session> | null;
    history: Partial<Session>[];
  }

  // Raise any v1 session to v2 shape. Pure — no side effects.
  function upgradeSession(s: Partial<Session> | null | undefined): Session | null {
    if (!s || typeof s !== "object") return null;
    const listing = s.listing
      ? {
          ...s.listing,
          motivoDaVenda: s.listing.motivoDaVenda ?? ("outros" as const),
        }
      : undefined;
    if (!listing) return null;

    return {
      id: s.id ?? `legacy-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`,
      listing,
      fipe: s.fipe ?? 0,
      targetPrice: s.targetPrice ?? 0,
      walkAwayPrice: s.walkAwayPrice ?? 0,
      messages: s.messages ?? [],
      round: s.round ?? 0,
      maxRounds: s.maxRounds ?? 6,
      status: s.status ?? "ended",
      startedAt: s.startedAt,
      endedAt: s.endedAt,
      endReason: s.endReason,
      // Phase 2 additions — defaults for migrated sessions:
      promptVersion: s.promptVersion ?? "v1.0.0",
      appliedConfig: s.appliedConfig ?? DEFAULT_AGENT_CONFIG,
      motivationScore: s.motivationScore,
      userRating: s.userRating,
      userNote: s.userNote,
      finalPrice: s.finalPrice,
      flags: s.flags,
    };
  }

  /**
   * Migrate Zustand persisted blob. Zustand passes the persisted *state* (not
   * the entire {state, version} envelope) to `migrate(persisted, version)`.
   * We accept both shapes defensively.
   */
  export function migrateV1ToV2(persisted: unknown): unknown {
    if (!persisted || typeof persisted !== "object") {
      return initState();
    }

    // Already v2 shape (idempotent)
    const candidate = persisted as { currentSession?: unknown; history?: unknown; agentConfig?: unknown };
    if (candidate.agentConfig && typeof candidate.agentConfig === "object") {
      return candidate;
    }

    // v1 shape — upgrade each session + inject agentConfig at top level
    const v1 = persisted as PersistedV1;
    const upgradedCurrent = upgradeSession(v1.currentSession ?? null);
    const upgradedHistory = Array.isArray(v1.history)
      ? v1.history.map((s) => upgradeSession(s)).filter((s): s is Session => s !== null)
      : [];

    return {
      currentSession: upgradedCurrent,
      history: upgradedHistory,
      agentConfig: DEFAULT_AGENT_CONFIG,
    };
  }

  import Anthropic from "@anthropic-ai/sdk";
  import type { MessageParam, TextBlockParam } from "@anthropic-ai/sdk/resources/messages";
  import type { StreamOptions } from "./types";

  const DEFAULT_MODEL = "claude-sonnet-4-6";

  /**
   * Anthropic adapter — Phase 2 extends Phase 1 with:
   *   1. cache_control on system when opts.cacheSystem=true
   *   2. few-shot pair prepend (from opts.fewShotPairs — may include cache_control on last assistant)
   *   3. onUsage telemetry callback after finalMessage()
   *
   * NOTE ABOUT TYPE CAST: opts.fewShotPairs is typed as LLMMessage[] (provider-agnostic).
   * The builder src/lib/prompts/few-shot.ts actually returns MessageParam[] with cache_control
   * embedded in the last assistant block. TypeScript narrows to LLMMessage[] via the interface
   * contract; at runtime the blocks carry their cache_control and Anthropic SDK accepts them.
   * The cast `as unknown as MessageParam[]` below is deliberate — see ADR note in PLAN 02-02.
   */
  export async function streamAnthropic(opts: StreamOptions): Promise<void> {
    const apiKey = process.env.ANTHROPIC_API_KEY;
    if (!apiKey) throw new Error("ANTHROPIC_API_KEY missing");
    const model = process.env.ANTHROPIC_MODEL ?? DEFAULT_MODEL;

    const client = new Anthropic({ apiKey });

    // Mutation 1: system prompt as TextBlockParam[] when caching enabled (AI-SPEC §4b.5)
    const system = opts.cacheSystem
      ? ([{ type: "text", text: opts.systemPrompt, cache_control: { type: "ephemeral" } }] satisfies TextBlockParam[])
      : opts.systemPrompt;

    // Mutation 2: prepend few-shot pairs to conversation history
    const fewShot = (opts.fewShotPairs ?? []) as unknown as MessageParam[];
    const messages: MessageParam[] = [
      ...fewShot,
      ...(opts.messages as unknown as MessageParam[]),
    ];

    const stream = client.messages.stream(
      {
        model,
        max_tokens: opts.maxTokens,
        system,
        messages,
      },
      { signal: opts.signal },
    );

    // Preserve Phase 1 contract: stream chunks untouched to caller (client strips tags)
    stream.on("text", (chunk: string) => opts.onText(chunk));

    const final = await stream.finalMessage();

    // Mutation 3: usage telemetry (AI-SPEC §7, §4b.5)
    opts.onUsage?.({
      input: final.usage.input_tokens,
      output: final.usage.output_tokens,
      cacheCreate: final.usage.cache_creation_input_tokens ?? 0,
      cacheRead: final.usage.cache_read_input_tokens ?? 0,
      stopReason: final.stop_reason,
    });
  }
  ```
