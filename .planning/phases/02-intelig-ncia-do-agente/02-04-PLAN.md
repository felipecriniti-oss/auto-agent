  "use client";

  import { Alert, AlertDescription } from "@/components/ui/alert";
  import { Button } from "@/components/ui/button";
  import {
    Form,
    FormControl,
    FormField,
    FormItem,
    FormLabel,
    FormMessage,
  } from "@/components/ui/form";
  import { Input } from "@/components/ui/input";
  import { Label } from "@/components/ui/label";
  import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";
  import { agentConfigSchema, type AgentConfig } from "@/lib/schemas/agent-config";
  import { useNegotiationStore } from "@/lib/stores/negotiation";
  import { zodResolver } from "@hookform/resolvers/zod";
  import { AlertCircle } from "lucide-react";
  import { useEffect } from "react";
  import { useForm } from "react-hook-form";
  import { toast } from "sonner";
  import { useShallow } from "zustand/react/shallow";

  export function AgentConfigForm() {
    const { agentConfig, setAgentConfig, isNegotiating } = useNegotiationStore(
      useShallow((s) => ({
        agentConfig: s.agentConfig,
        setAgentConfig: s.setAgentConfig,
        isNegotiating: s.currentSession?.status === "negotiating",
      })),
    );

    const form = useForm<AgentConfig>({
      resolver: zodResolver(agentConfigSchema),
      defaultValues: agentConfig,
      mode: "onBlur",
    });

    // Sync form when store agentConfig changes externally (e.g., after migration)
    useEffect(() => {
      form.reset(agentConfig);
      // Eslint exhaustive-deps: `form` is stable via useForm — safe to omit
      // eslint-disable-next-line react-hooks/exhaustive-deps
    }, [agentConfig]);

    function onSubmit(data: AgentConfig) {
      setAgentConfig(data);
      toast.success("Configuração salva.");
    }

    return (
      <Form {...form}>
        <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4 p-4">
          <FormField
            control={form.control}
            name="targetDiscount"
            render={({ field }) => (
              <FormItem>
                <FormLabel htmlFor="targetDiscount">Desconto-alvo (%)</FormLabel>
                <FormControl>
                  <Input
                    id="targetDiscount"
                    type="number"
                    min={5}
                    max={40}
                    step={1}
                    value={Math.round(field.value * 100)}
                    onChange={(e) => field.onChange(Number(e.target.value) / 100)}
                    onBlur={field.onBlur}
                    data-testid="config-targetDiscount"
                  />
                </FormControl>
                <FormMessage />
              </FormItem>
            )}
          />

          <FormField
            control={form.control}
            name="maxRounds"
            render={({ field }) => (
              <FormItem>
                <FormLabel htmlFor="maxRounds">Máximo de rodadas</FormLabel>
                <FormControl>
                  <Input
                    id="maxRounds"
                    type="number"
                    min={2}
                    max={12}
                    step={1}
                    value={field.value}
                    onChange={(e) => field.onChange(Number(e.target.value))}
                    onBlur={field.onBlur}
                    data-testid="config-maxRounds"
                  />
                </FormControl>
                <FormMessage />
              </FormItem>
            )}
          />

          <FormField
            control={form.control}
            name="tone"
            render={({ field }) => (
              <FormItem>
                <FormLabel>Tom do agente</FormLabel>
                <FormControl>
                  <RadioGroup
                    value={field.value}
                    onValueChange={field.onChange}
                    className="flex gap-4"
                    data-testid="config-tone"
