  ```typescript
  import { describe, expect, it } from "vitest";
  import Anthropic from "@anthropic-ai/sdk";
  import type { MessageParam, TextBlockParam } from "@anthropic-ai/sdk/resources/messages";
  import { buildSystemPromptV2 } from "../system-v2";
  import { buildFewShot } from "../few-shot";
  import { COMPARABLES, formatComparablesForPrompt } from "@/lib/data/comparables";
  import { computeMotivationScore } from "@/lib/scoring/motivation";
  import type { Listing } from "@/lib/schemas/listing";
  import type { AgentConfig } from "@/lib/schemas/agent-config";

  const HAS_KEY = !!process.env.ANTHROPIC_API_KEY;

  // Sample payload representative of Felipe's typical demo — medium complexity
  const sampleListing: Listing = {
    marca: "Honda", modelo: "Civic EXL", ano: 2019, km: 45000,
    precoPedido: 85000, cidade: "São Paulo", diasOnline: 30, reducoes: 1,
    motivoDaVenda: "outros",
  };
  const sampleConfig: AgentConfig = {
    targetDiscount: 0.25, maxRounds: 6, tone: "casual", initialAnchorStrategy: "moderado",
  };

  describe.skipIf(!HAS_KEY)("prompt size measurement — live Anthropic countTokens", () => {
    it("measures SYSTEM_PROMPT_V2 + few-shot total tokens for cache eligibility", async () => {
      const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

      const motivation = computeMotivationScore(sampleListing);
      const comparablesText = formatComparablesForPrompt(COMPARABLES);
      const systemPrompt = buildSystemPromptV2(
        sampleListing, 82000, 61500, 73800, sampleConfig, motivation, comparablesText,
      );

      const systemBlocks: TextBlockParam[] = [
        { type: "text", text: systemPrompt, cache_control: { type: "ephemeral" } },
      ];
      const messages: MessageParam[] = [
        ...buildFewShot(),
        { role: "user", content: "Olá, vi seu anúncio." },
      ];

      const result = await client.messages.countTokens({
        model: "claude-sonnet-4-6",
        system: systemBlocks,
        messages,
      });

      console.log(
        JSON.stringify({
          scope: "prompt_size_measurement",
          systemPromptChars: systemPrompt.length,
          totalInputTokens: result.input_tokens,
          cacheThreshold: 2048,
          meetsCacheThreshold: result.input_tokens >= 2048,
        }),
      );

      // Soft expectation — we WANT it >= 2048 for caching to kick in;
      // if it's not, test still passes but logs the gap for planner decision (Task 1 SUMMARY).
      expect(result.input_tokens).toBeGreaterThan(800); // Sanity: prompt is non-trivially sized
      // If meetsCacheThreshold false, planner in SUMMARY documents:
      //   - Accept no-caching (cost impact documented in AI-SPEC §4b.5 fallback: $2.36 vs $1.98 — both under R$ 50 cap)
      //   - OR remove cacheSystem=true from route handler to avoid write-cost overhead
      //   - OR inline few-shot into system prompt to bump size (adds ~900 tokens)
    }, 30_000);

    it("measures system-only (no few-shot) tokens for baseline comparison", async () => {
      const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

      const motivation = computeMotivationScore(sampleListing);
      const comparablesText = formatComparablesForPrompt(COMPARABLES);
      const systemPrompt = buildSystemPromptV2(
        sampleListing, 82000, 61500, 73800, sampleConfig, motivation, comparablesText,
      );

      const result = await client.messages.countTokens({
        model: "claude-sonnet-4-6",
        system: systemPrompt,
        messages: [{ role: "user", content: "Olá." }],
      });

      console.log(
        JSON.stringify({
          scope: "prompt_size_measurement.system_only",
          systemOnlyTokens: result.input_tokens,
        }),
      );

      expect(result.input_tokens).toBeGreaterThan(500);
    }, 30_000);
  });

  describe("prompt size measurement — offline fallback (no API key)", () => {
    it("logs char length as proxy when no ANTHROPIC_API_KEY (skipped in CI)", () => {
      if (HAS_KEY) return; // live test covers this case
      const motivation = computeMotivationScore(sampleListing);
      const comparablesText = formatComparablesForPrompt(COMPARABLES);
      const systemPrompt = buildSystemPromptV2(
        sampleListing, 82000, 61500, 73800, sampleConfig, motivation, comparablesText,
      );
      // Rough heuristic for pt-BR: ~4 chars/token — sanity proxy only
      const approxTokens = Math.ceil(systemPrompt.length / 4);
      console.log(
        JSON.stringify({
          scope: "prompt_size_offline_fallback",
          systemPromptChars: systemPrompt.length,
          approxTokens,
          note: "Run with ANTHROPIC_API_KEY for empirical count",
        }),
      );
      expect(systemPrompt.length).toBeGreaterThan(2000); // Template is meaningfully sized
    });
  });
  ```
